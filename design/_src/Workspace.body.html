from typing import Iterable, Optional
 
 
def first_duplicate(stream: Iterable[int]) -> Optional[int]:
    """Devuelve el primer valor repetido del stream."""
    seen = set()
    for item in stream:
        if item in seen:
            return item
        seen.add(item)
    return None
